Simple Bayesian Network Implementation

Example Used: Student Performance Prediction

Variables:
Study
Attendance
Exam
Result

Query Example:
P(Result | Study=True, Attendance=True)

Run:
    python bayesian_network.py
"""


class BayesianNetwork:

    def __init__(self):

        self.variables = []
        self.parents = {}
        self.cpt = {}

    def add_variable(self, variable, parents, cpt):

        self.variables.append(variable)
        self.parents[variable] = parents
        self.cpt[variable] = cpt

    def probability(self, variable, value, evidence):

        parent_nodes = self.parents[variable]

        if len(parent_nodes) == 0:

            true_probability = self.cpt[variable][()]

        else:

            key = tuple(evidence[parent] for parent in parent_nodes)

            true_probability = self.cpt[variable][key]

        if value == True:
            return true_probability

        return 1 - true_probability


def enumerate_all(variables, evidence, network):

    if len(variables) == 0:
        return 1.0

    first = variables[0]

    remaining = variables[1:]

    if first in evidence:

        probability = network.probability(
            first,
            evidence[first],
            evidence
        )

        return probability * enumerate_all(
            remaining,
            evidence,
            network
        )

    total = 0

    for value in [True, False]:

        new_evidence = evidence.copy()

        new_evidence[first] = value

        probability = network.probability(
            first,
            value,
            new_evidence
        )

        total += probability * enumerate_all(
            remaining,
            new_evidence,
            network
        )

    return total


def inference(query_variable, evidence, network):

    distribution = {}

    for value in [True, False]:

        new_evidence = evidence.copy()

        new_evidence[query_variable] = value

        distribution[value] = enumerate_all(
            network.variables,
            new_evidence,
            network
        )

    total = distribution[True] + distribution[False]

    distribution[True] = distribution[True] / total
    distribution[False] = distribution[False] / total

    return distribution


def build_student_network():

    network = BayesianNetwork()

    # STUDY

    network.add_variable(
        "Study",
        [],
        {
            (): 0.70
        }
    )

    # ATTENDANCE

    network.add_variable(
        "Attendance",
        [],
        {
            (): 0.80
        }
    )

    # EXAM

    network.add_variable(
        "Exam",
        ["Study", "Attendance"],
        {
            (True, True): 0.95,
            (True, False): 0.75,
            (False, True): 0.60,
            (False, False): 0.20
        }
    )

    # RESULT

    network.add_variable(
        "Result",
        ["Exam"],
        {
            (True,): 0.90,
            (False,): 0.30
        }
    )

    return network


def main():

    network = build_student_network()

    evidence = {

        "Study": True,
        "Attendance": True
    }

    result = inference(
        "Result",
        evidence,
        network
    )

    print("STUDENT BAYESIAN NETWORK")
    print("Query: P(Result | Study=True, Attendance=True)")

    print("P(Result=True):",
          round(result[True], 5))

    print("P(Result=False):",
          round(result[False], 5))


if __name__ == "__main__":
    main()
